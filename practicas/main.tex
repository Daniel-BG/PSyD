% --- Comando Base para "Burbujas/Badges" ---
\newcommand{\codebubble}[3]{%
  \tcbox[
    on line,
    arc=4pt,
    outer arc=4pt,
    colback=#1,
    colframe=#1,
    coltext=#2,
    boxsep=0pt, left=5pt, right=5pt, top=2pt, bottom=2pt,
    boxrule=0pt
  ]{\small\texttt{#3}}%
}
